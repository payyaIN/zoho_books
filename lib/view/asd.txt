String getInvoiceStatusText(int? isVerified, int? status) {
  if (status == 2) return "REJECTED";
  if (isVerified == 1) return "APPROVED";
  return "PENDING";
}

String getInvoiceStatusImage(int? isVerified, int? status) {
  if (status == 2) return AppImages.rejectedBox;
  if (isVerified == 1) return AppImages.approvedBox;
  return AppImages.pendingBox;
}

Stack billData({
  required BuildContext context,
  required bool isBill,
  required int? isVerified,
  required int? invoiceStatus,
  required String rightText1,
  required String rightText2,
  required String rightText3,
  required String rightText4,
}) {
  var width = MediaQuery.of(context).size.width;
  
  // Get the status text and image
  final statusText = getInvoiceStatusText(isVerified, invoiceStatus);
  final statusImage = getInvoiceStatusImage(isVerified, invoiceStatus);
  
  return Stack(
    children: [
      ReusableContainer(
        color: AppColors.appFullWhite,
        width: 344,
        child: Stack(
          children: [
            Padding(
              padding: const EdgeInsets.only(
                  left: 14, top: 24, right: 107, bottom: 24),
              child: ReusableContainer(
                width: width,
                child: Transform.translate(
                  offset: const Offset(0, 0),
                  child: Column(
                    mainAxisAlignment: MainAxisAlignment.spaceBetween,
                    crossAxisAlignment: CrossAxisAlignment.start,
                    children: [
                      rowText2(
                          leftText: isBill ? "Bill#" : "Invoice#",
                          rightText: rightText1,
                          is60015: false,
                          is70016: false,
                          isQuantity: false,
                          is70020: false),
                      GapSpace.height15,
                      rowText2(
                        leftText: "Date",
                        rightText: rightText2,
                        is60015: false,
                        is70016: false,
                        is70020: false,
                        isQuantity: false,
                      ),
                      GapSpace.height15,
                      rowText2(
                        leftText: "Due Date",
                        rightText: rightText3,
                        is60015: false,
                        is70016: false,
                        is70020: false,
                        isQuantity: false,
                      ),
                      GapSpace.height15,
                      rowText2(
                        leftText: isBill ? "Branch" : "Currency",
                        rightText: rightText4,
                        is60015: false,
                        is70016: false,
                        is70020: false,
                        isQuantity: false,
                      ),
                    ],
                  ),
                ),
              ),
            ),
            Positioned(
                right: -20,
                child: Stack(
                  children: [
                    SvgPictureWidget(image: statusImage, height: 21, width: 128),
                    Transform.translate(
                      offset: Offset(40, 2),
                      child: ReusableText(
                        text: statusText,
                        color: AppColors.appWhiteColor,
                        fontSize: 11,
                        fontWeight: FontWeight.w500,
                      ),
                    )
                  ],
                )),
          ],
        ),
      ),
    ],
  );
}
